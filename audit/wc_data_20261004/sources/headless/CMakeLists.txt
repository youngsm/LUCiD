cmake_minimum_required(VERSION 3.16...3.30)
project(PhotonSimAudit)
find_package(Geant4 REQUIRED)
find_package(ROOT REQUIRED COMPONENTS Core Tree)
set(photon_source "${CMAKE_CURRENT_SOURCE_DIR}/../PhotonSim-v1.0.0")
file(GLOB photon_sources "${photon_source}/src/*.cc")
add_executable(PhotonSim "${photon_source}/PhotonSim.cc" ${photon_sources})
target_include_directories(PhotonSim PRIVATE "${photon_source}/include" ${ROOT_INCLUDE_DIRS})
target_link_libraries(PhotonSim PRIVATE ${Geant4_LIBRARIES} ${ROOT_LIBRARIES})
